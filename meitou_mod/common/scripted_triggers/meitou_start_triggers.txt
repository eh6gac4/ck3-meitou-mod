# Vanilla seeds these four swords when a 1066 or later bookmark begins.
# Keep the 867 campaign eligible even after its calendar reaches 1066.
meitou_early_campaign_trigger = {
    OR = {
        current_date < 1066.9.15
        has_global_variable = meitou_early_campaign
    }
}

meitou_higekiri_available_trigger = {
    meitou_early_campaign_trigger = yes
    NOT = { has_global_variable = meitou_higekiri_found }
}

meitou_hizamaru_available_trigger = {
    meitou_early_campaign_trigger = yes
    NOT = { has_global_variable = meitou_hizamaru_found }
}

meitou_mikazuki_available_trigger = {
    meitou_early_campaign_trigger = yes
    NOT = { has_global_variable = meitou_mikazuki_found }
}

meitou_dojigiri_available_trigger = {
    meitou_early_campaign_trigger = yes
    NOT = { has_global_variable = meitou_dojigiri_found }
}

# The base game seeds Odenta in campaigns begun on or after 1178.10.1.
# A campaign begun earlier can still discover it after that date.
meitou_odenta_available_trigger = {
    OR = {
        current_date < 1178.10.1
        has_global_variable = meitou_odenta_unseeded_campaign
    }
    NOT = { has_global_variable = meitou_odenta_found }
}

meitou_juzumaru_available_trigger = {
    NOT = { has_global_variable = meitou_juzumaru_found }
}

meitou_onimaru_available_trigger = {
    NOT = { has_global_variable = meitou_onimaru_found }
}
