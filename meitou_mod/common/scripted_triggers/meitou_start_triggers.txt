# Vanilla seeds these four swords when a 1066 or later bookmark begins.
# Keep the 867 campaign eligible even after its calendar reaches 1066.
meitou_early_campaign_trigger = {
    OR = {
        current_date < 1066.9.15
        has_global_variable = meitou_early_campaign
    }
}

meitou_higekiri_available_trigger = {
    meitou_early_campaign_trigger = yes
    NOT = { has_global_variable = meitou_higekiri_found }
}

meitou_hizamaru_available_trigger = {
    meitou_early_campaign_trigger = yes
    NOT = { has_global_variable = meitou_hizamaru_found }
}

meitou_mikazuki_available_trigger = {
    meitou_early_campaign_trigger = yes
    NOT = { has_global_variable = meitou_mikazuki_found }
}

meitou_dojigiri_available_trigger = {
    meitou_early_campaign_trigger = yes
    NOT = { has_global_variable = meitou_dojigiri_found }
}

# The base game seeds Odenta in campaigns begun on or after 1178.10.1.
# A campaign begun earlier can still discover it after that date.
meitou_odenta_available_trigger = {
    OR = {
        current_date < 1178.10.1
        has_global_variable = meitou_odenta_unseeded_campaign
    }
    NOT = { has_global_variable = meitou_odenta_found }
}

meitou_juzumaru_available_trigger = {
    NOT = { has_global_variable = meitou_juzumaru_found }
}

meitou_onimaru_available_trigger = {
    NOT = { has_global_variable = meitou_onimaru_found }
}

# Artifact scope. Match the base modifiers so vanilla-seeded swords are eligible too.
meitou_named_sword_artifact_trigger = {
    OR = {
        has_artifact_modifier = higekiri_sword_modifier
        has_artifact_modifier = hizamaru_sword_modifier
        has_artifact_modifier = mikazuki_munechika_sword_modifier
        has_artifact_modifier = dojigiri_sword_modifier
        has_artifact_modifier = odenta_sword_modifier
        has_artifact_modifier = meitou_higekiri_modifier
        has_artifact_modifier = meitou_hizamaru_modifier
        has_artifact_modifier = meitou_mikazuki_modifier
        has_artifact_modifier = meitou_dojigiri_modifier
        has_artifact_modifier = meitou_ookanehira_modifier
        has_artifact_modifier = meitou_juzumaru_modifier
        has_artifact_modifier = meitou_onimaru_modifier
    }
}

# Character scope. These are the CK3 baronies Kyo, Oku, Kuboya, and Mii.
meitou_swordsmith_destination_trigger = {
    OR = {
        location = province:10603 # b_kyo, Yamashiro
        location = province:10587 # b_oku, Bizen
        location = province:10583 # b_kuboya, Bitchu
        location = province:10533 # b_mii, Chikugo
    }
}

# Completed blades no longer participate in workshop selection.
meitou_upgrade_eligible_artifact_trigger = {
    meitou_named_sword_artifact_trigger = yes
    NOT = { meitou_polish_3_trigger = yes }
}

# Character scope; the selected artifact determines the payment.
meitou_can_pay_polish_1_trigger = {
    gold >= 70
    exists = scope:meitou_upgrade_target
    scope:meitou_upgrade_target = { NOT = { meitou_polish_1_trigger = yes } NOT = { meitou_polish_2_trigger = yes } NOT = { meitou_polish_3_trigger = yes } }
}
