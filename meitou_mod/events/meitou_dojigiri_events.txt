# History-inspired fiction; see README for sources and mechanics.

meitou_travel.0004 = {
    type = character_event
    title = meitou_travel.0004.t
    desc = meitou_travel.0004.desc
    theme = travel
    override_background = { reference = wilderness_forest }
    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        character = scope:meitou_dojigiri_guardian
        animation = personality_honorable
    }
    trigger = {
        is_ai = no
        meitou_dojigiri_available_trigger = yes
        OR = {
            AND = {
                is_available_travelling_adult = yes
                location = { geographical_region = world_asia_japan }
                NOT = { has_character_flag = meitou_dojigiri_recent_encounter }
            }
            AND = {
                debug_only = yes
                has_character_flag = meitou_debug_dojigiri_intro
            }
        }
    }
    immediate = {
        if = {
            limit = { NOT = { exists = scope:meitou_dojigiri_guardian } }
            create_character = {
                template = pool_repopulate_martial
                age = { 35 60 }
                location = root.location
                culture = root.location.culture
                faith = root.location.faith
                gender_female_chance = 50
                save_scope_as = meitou_dojigiri_guardian
                after_creation = {
                    add_character_flag = created
                    add_character_flag = { flag = silently_disappear_me years = 1 }
                }
            }
        }
        remove_character_flag = meitou_debug_dojigiri_intro
        add_character_flag = { flag = meitou_dojigiri_recent_encounter days = 180 }
    }
    option = {
        name = meitou_travel.0004.a
        trigger_event = { id = meitou_travel.0104 }
    }
    option = {
        name = meitou_travel.leave
        if = {
            limit = { exists = scope:meitou_dojigiri_guardian }
            scope:meitou_dojigiri_guardian = { silent_disappearance_effect = yes }
        }
    }
}

        name = meitou_travel.withdraw
        trigger = { meitou_dojigiri_available_trigger = yes }
        if = {
            limit = { exists = scope:meitou_dojigiri_guardian }
            scope:meitou_dojigiri_guardian = { silent_disappearance_effect = yes }
        }
    }
    option = {
        name = meitou_travel.unavailable
        trigger = { NOT = { meitou_dojigiri_available_trigger = yes } }
        trigger_event = { id = meitou_travel.0999 }
    }
    option = {
        name = meitou_debug.force_success
