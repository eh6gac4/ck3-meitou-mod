# History-inspired fiction; see README for sources and mechanics.

meitou_travel.0001 = {
    type = character_event
    title = meitou_travel.0001.t
    desc = meitou_travel.0001.desc
    theme = travel
    override_background = { reference = ep2_travel_bridge }
    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        character = scope:meitou_higekiri_guardian
        animation = personality_honorable
    }
    trigger = {
        is_ai = no
        meitou_higekiri_available_trigger = yes
        OR = {
            AND = {
                is_available_travelling_adult = yes
                location = { geographical_region = world_asia_japan }
                NOT = { has_character_flag = meitou_higekiri_recent_encounter }
            }
            AND = {
                debug_only = yes
                has_character_flag = meitou_debug_higekiri_intro
            }
        }
    }
    immediate = {
        if = {
            limit = { NOT = { exists = scope:meitou_higekiri_guardian } }
            create_character = {
                template = pool_repopulate_martial
                age = { 60 70 }
                location = root.location
                culture = root.location.culture
                faith = root.location.faith
                gender_female_chance = 0
                save_scope_as = meitou_higekiri_guardian
                after_creation = {
                    add_character_flag = created
                    add_character_flag = { flag = silently_disappear_me years = 1 }
                }
            }
        }
        remove_character_flag = meitou_debug_higekiri_intro
        add_character_flag = { flag = meitou_higekiri_recent_encounter days = 180 }
    }
    option = {
        name = meitou_travel.0001.a
        trigger_event = { id = meitou_travel.0101 }
    }
    option = {
        name = meitou_travel.leave
        if = {
            limit = { exists = scope:meitou_higekiri_guardian }
            scope:meitou_higekiri_guardian = { silent_disappearance_effect = yes }
        }
    }
}
