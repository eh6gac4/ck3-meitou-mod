meitou_travel.0002 = {
    type = character_event
    title = meitou_travel.0002.t
    desc = meitou_travel.0002.desc
    theme = travel
    override_background = { reference = tgp_crossroad_inn_asia }
    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        character = scope:meitou_hizamaru_guardian
        animation = personality_honorable
    }
    trigger = {
        is_ai = no
        meitou_hizamaru_available_trigger = yes
        OR = {
            AND = {
                is_available_travelling_adult = yes
                location = { geographical_region = world_asia_japan }
                NOT = { has_character_flag = meitou_hizamaru_recent_encounter }
            }
            AND = {
                debug_only = yes
                has_character_flag = meitou_debug_hizamaru_intro
            }
        }
    }
    immediate = {
        if = {
            limit = { NOT = { exists = scope:meitou_hizamaru_guardian } }
            create_character = {
                template = pool_repopulate_learning
                age = { 20 35 }
                location = root.location
                culture = root.location.culture
                faith = root.location.faith
                gender_female_chance = 50
                save_scope_as = meitou_hizamaru_guardian
                after_creation = {
                    add_character_flag = created
                    add_character_flag = { flag = silently_disappear_me years = 1 }
                }
            }
        }
        remove_character_flag = meitou_debug_hizamaru_intro
        add_character_flag = { flag = meitou_hizamaru_recent_encounter days = 180 }
    }
    option = {
        name = meitou_travel.0002.a
        trigger_event = { id = meitou_travel.0102 }
    }
    option = {
        name = meitou_travel.leave
        if = {
            limit = { exists = scope:meitou_hizamaru_guardian }
            scope:meitou_hizamaru_guardian = { silent_disappearance_effect = yes }
        }
    }
}
