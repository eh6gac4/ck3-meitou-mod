# History-inspired fiction; see README for sources and mechanics.

meitou_travel.0003 = {
    type = character_event
    title = meitou_travel.0003.t
    desc = meitou_travel.0003.desc
    theme = travel
    override_background = { reference = tgp_relaxing_room_asia }
    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        character = scope:meitou_mikazuki_guardian
        animation = personality_honorable
    }
    trigger = {
        is_ai = no
        meitou_mikazuki_available_trigger = yes
        OR = {
            AND = {
                is_available_travelling_adult = yes
                location = { geographical_region = world_asia_japan }
                NOT = { has_character_flag = meitou_mikazuki_recent_encounter }
            }
            AND = {
                debug_only = yes
                has_character_flag = meitou_debug_mikazuki_intro
            }
        }
    }
    immediate = {
        if = {
            limit = { NOT = { exists = scope:meitou_mikazuki_guardian } }
            create_character = {
                template = pool_repopulate_diplomacy
                age = { 35 60 }
                location = root.location
                culture = root.location.culture
                faith = root.location.faith
                gender_female_chance = 0
                save_scope_as = meitou_mikazuki_guardian
                after_creation = {
                    add_character_flag = created
                    add_character_flag = { flag = silently_disappear_me years = 1 }
                }
            }
        }
        remove_character_flag = meitou_debug_mikazuki_intro
        add_character_flag = { flag = meitou_mikazuki_recent_encounter days = 180 }
    }
    option = {
        name = meitou_travel.0003.a
        trigger_event = { id = meitou_travel.0103 }
    }
    option = {
        name = meitou_travel.leave
        if = {
            limit = { exists = scope:meitou_mikazuki_guardian }
            scope:meitou_mikazuki_guardian = { silent_disappearance_effect = yes }
        }
    }
}

                        trigger_event = { id = meitou_travel.0303 }

                }
            }

        else = { trigger_event = { id = meitou_travel.0999 } }
    }

        name = meitou_travel.0103.c
        trigger = { meitou_mikazuki_available_trigger = yes gold >= 80 }
        if = {
            limit = { meitou_mikazuki_available_trigger = yes gold >= 80 }
            remove_short_term_gold = 80
            create_artifact_mikazuki_munechika_sword_effect = { OWNER = root }
                        set_global_variable = meitou_mikazuki_found
                        trigger_event = { id = meitou_travel.0203 }
        }
        else = { trigger_event = { id = meitou_travel.0999 } }
    }
    option = {
        name = meitou_travel.withdraw
        trigger = { meitou_mikazuki_available_trigger = yes }
        if = {
            limit = { exists = scope:meitou_mikazuki_guardian }
            scope:meitou_mikazuki_guardian = { silent_disappearance_effect = yes }
        }
    }
    option = {
        name = meitou_travel.unavailable
        trigger = { NOT = { meitou_mikazuki_available_trigger = yes } }
        trigger_event = { id = meitou_travel.0999 }
    }
    option = {
        name = meitou_debug.force_success

            limit = { debug_only = yes is_ai = no meitou_mikazuki_available_trigger = yes }
            add_stress = 15
            trigger_event = { id = meitou_travel.0303 }
        }
    }
