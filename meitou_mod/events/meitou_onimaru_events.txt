# History-inspired fiction; see README for sources and mechanics.

meitou_travel.0008 = {
    type = character_event
    title = meitou_travel.0008.t
    desc = meitou_travel.0008.desc
    theme = travel
    override_background = { reference = wilderness_forest }
    left_portrait = {
        character = root
        animation = personality_rational
    }
    right_portrait = {
        character = scope:meitou_onimaru_guardian
        animation = personality_honorable
    }
    trigger = {
        is_ai = no
        meitou_onimaru_available_trigger = yes
        OR = {
            AND = {
                is_available_travelling_adult = yes
                location = { geographical_region = world_asia_japan }
                NOT = { has_character_flag = meitou_onimaru_recent_encounter }
            }
            AND = {
                debug_only = yes
                has_character_flag = meitou_debug_onimaru_intro
            }
        }
    }
    immediate = {
        if = {
            limit = { NOT = { exists = scope:meitou_onimaru_guardian } }
            create_character = {
                template = pool_repopulate_martial
                age = { 35 60 }
                location = root.location
                culture = root.location.culture
                faith = root.location.faith
                gender_female_chance = 50
                save_scope_as = meitou_onimaru_guardian
                after_creation = {
                    add_character_flag = created
                    add_character_flag = { flag = silently_disappear_me years = 1 }
                }
            }
        }
        remove_character_flag = meitou_debug_onimaru_intro
        add_character_flag = { flag = meitou_onimaru_recent_encounter days = 180 }
    }
    option = {
        name = meitou_travel.0008.a
        trigger_event = { id = meitou_travel.0108 }
    }
    option = {
        name = meitou_travel.leave
        if = {
            limit = { exists = scope:meitou_onimaru_guardian }
            scope:meitou_onimaru_guardian = { silent_disappearance_effect = yes }
        }
    }
}

                        meitou_create_onimaru_effect = { OWNER = root }

                    }
                }
            }
        }
        else = { trigger_event = { id = meitou_travel.0999 } }
    }
    option = {
        name = meitou_travel.0108.b
        trigger = { meitou_onimaru_available_trigger = yes }
        if = {
            limit = { meitou_onimaru_available_trigger = yes }

                random_list = {
                    80 = {

                        set_global_variable = meitou_onimaru_found
                        trigger_event = { id = meitou_travel.0208 }
                    }
                    55 = {
                        add_stress = 25
                        trigger_event = { id = meitou_travel.0308 }
                    }
                }
            }
        }
        else = { trigger_event = { id = meitou_travel.0999 } }
    }
    option = {
        name = meitou_travel.0108.c
        trigger = { meitou_onimaru_available_trigger = yes gold >= 70 }
        if = {
            limit = { meitou_onimaru_available_trigger = yes gold >= 70 }
            remove_short_term_gold = 70
            meitou_create_onimaru_effect = { OWNER = root }
                        set_global_variable = meitou_onimaru_found
                        trigger_event = { id = meitou_travel.0208 }
        }
        else = { trigger_event = { id = meitou_travel.0999 } }
    }
    option = {
        name = meitou_travel.withdraw
        trigger = { meitou_onimaru_available_trigger = yes }
        if = {
            limit = { exists = scope:meitou_onimaru_guardian }
            scope:meitou_onimaru_guardian = { silent_disappearance_effect = yes }
        }
    }
    option = {
        name = meitou_travel.unavailable
        trigger = { NOT = { meitou_onimaru_available_trigger = yes } }
        trigger_event = { id = meitou_travel.0999 }
    }
    option = {
        name = meitou_debug.force_success
        trigger = { debug_only = yes is_ai = no meitou_onimaru_available_trigger = yes }
        if = {
            limit = { debug_only = yes is_ai = no meitou_onimaru_available_trigger = yes }
            meitou_create_onimaru_effect = { OWNER = root }
            set_global_variable = meitou_onimaru_found

    }

        name = meitou_debug.force_failure

            limit = { debug_only = yes is_ai = no meitou_onimaru_available_trigger = yes }
            add_stress = 25
            trigger_event = { id = meitou_travel.0308 }
        }
    }

}
